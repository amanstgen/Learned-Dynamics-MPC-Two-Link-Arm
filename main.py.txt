import pybullet as p
import pybullet_data
import numpy as np
import torch
import torch.nn as nn
import torch.optim as optim
import matplotlib.pyplot as plt
import os
import zipfile

# Create output directory for figures
os.makedirs("figures", exist_ok=True)

# ==========================================
# 1. NEURAL NETWORK DEFINITION
# ==========================================
class MLP(nn.Module):
    def __init__(self):
        super(MLP, self).__init__()
        self.net = nn.Sequential(
            nn.Linear(6, 64), nn.ReLU(),
            nn.Linear(64, 64), nn.ReLU(),
            nn.Linear(64, 64), nn.ReLU(),
            nn.Linear(64, 4)
        )
    def forward(self, x):
        return self.net(x)

# ==========================================
# 2. PYBULLET ENVIRONMENT SETUP
# ==========================================
def setup_sim(friction_level=1.0):
    p.connect(p.DIRECT) 
    p.setAdditionalSearchPath(pybullet_data.getDataPath())
    p.setGravity(0, 0, 0)
    
    # We create the URDF on the fly
    urdf_string = """
    <robot name="two_link">
      <link name="base_link"><inertial><mass value="1.0"/><inertia ixx="0.1" iyy="0.1" izz="0.1" ixy="0" ixz="0" iyz="0"/></inertial><visual><geometry><box size="0.1 0.1 0.1"/></geometry></visual></link>
      <link name="link1"><inertial><origin xyz="0.5 0 0"/><mass value="1.0"/><inertia ixx="0.1" iyy="0.1" izz="0.1" ixy="0" ixz="0" iyz="0"/></inertial><visual><origin xyz="0.5 0 0"/><geometry><box size="1.0 0.05 0.05"/></geometry></visual></link>
      <joint name="joint1" type="revolute"><parent link="base_link"/><child link="link1"/><origin xyz="0 0 0"/><axis xyz="0 0 1"/><limit lower="-3.14" upper="3.14" effort="100" velocity="10"/></joint>
      <link name="link2"><inertial><origin xyz="0.4 0 0"/><mass value="0.8"/><inertia ixx="0.1" iyy="0.1" izz="0.1" ixy="0" ixz="0" iyz="0"/></inertial><visual><origin xyz="0.4 0 0"/><geometry><box size="0.8 0.05 0.05"/></geometry></visual></link>
      <joint name="joint2" type="revolute"><parent link="link1"/><child link="link2"/><origin xyz="1.0 0 0"/><axis xyz="0 0 1"/><limit lower="-3.14" upper="3.14" effort="100" velocity="10"/></joint>
    </robot>
    """
    urdf_path = "two_link.urdf"
    with open(urdf_path, "w") as f:
        f.write(urdf_string)
    
    arm = p.loadURDF(urdf_path, basePosition=[0,0,0], useFixedBase=True)
    
    for j in [0, 1]:
        p.changeDynamics(arm, j, jointDamping=friction_level * 2.0)
        p.setJointMotorControl2(arm, j, p.VELOCITY_CONTROL, force=0) 
    p.setTimeStep(1/240.)
    return arm

def get_state(arm):
    js = p.getJointStates(arm, [0, 1])
    return np.array([js[0][0], js[1][0], js[0][1], js[1][1]])

def step_sim(arm, torque, steps=12):
    p.setJointMotorControlArray(arm, [0, 1], p.TORQUE_CONTROL, forces=torque)
    for _ in range(steps): p.stepSimulation()
    return get_state(arm)

# ==========================================
# 3. DATA GENERATION & TRAINING
# ==========================================
print("Generating Data...")
X_data, Y_data = [], []
arm_data = setup_sim(friction_level=1.0)

for ep in range(50): # Reduced episodes for quick run
    p.resetJointState(arm_data, 0, np.random.uniform(-1.0, 1.0), targetVelocity=0)
    p.resetJointState(arm_data, 1, np.random.uniform(-1.0, 1.0), targetVelocity=0)
    
    for step in range(50):
        state = get_state(arm_data)
        torque = np.random.uniform(-50, 50, 2) 
        next_state = step_sim(arm_data, torque, steps=12)
        X_data.append(np.concatenate([state, torque]))
        Y_data.append(next_state - state) 

X_data = np.array(X_data, dtype=np.float32)
Y_data = np.array(Y_data, dtype=np.float32)

X_mean, X_std = X_data.mean(axis=0), X_data.std(axis=0) + 1e-8
Y_mean, Y_std = Y_data.mean(axis=0), Y_data.std(axis=0) + 1e-8

split_train = int(len(X_data) * 0.8)
X_train = (X_data[:split_train] - X_mean) / X_std
Y_train = (Y_data[:split_train] - Y_mean) / Y_std
X_val = (X_data[split_train:] - X_mean) / X_std
Y_val = (Y_data[split_train:] - Y_mean) / Y_std

print("Training Neural Network...")
device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
model = MLP().to(device)
criterion = nn.MSELoss()
optimizer = optim.Adam(model.parameters(), lr=0.001)

X_t, Y_t = torch.tensor(X_train).to(device), torch.tensor(Y_train).to(device)
X_v, Y_v = torch.tensor(X_val).to(device), torch.tensor(Y_val).to(device)

train_losses, val_losses = [], []
for epoch in range(50):
    model.train()
    optimizer.zero_grad()
    loss = criterion(model(X_t), Y_t)
    loss.backward()
    optimizer.step()
    train_losses.append(loss.item())
    
    model.eval()
    with torch.no_grad():
        val_losses.append(criterion(model(X_v), Y_v).item())

# ==========================================
# 4. GENERATE PUBLICATION FIGURES
# ==========================================
print("Generating Figures...")

# Fig 6.1: Training Loss
plt.figure()
epochs = np.arange(50)
train_curve = np.array(train_losses)
val_curve = np.array(val_losses)
plt.plot(epochs, train_curve, label='Train Loss')
plt.plot(epochs, val_curve, label='Val Loss', alpha=0.8)
plt.xlabel('Epoch'); plt.ylabel('MSE Loss'); plt.legend()
plt.title('Model Training Convergence')
plt.savefig('figures/fig6_1_loss.png', dpi=300); plt.close()
